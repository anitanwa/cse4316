\section{Introduction}
This SRS specifies \productname, a web application for community service requests.
It follows ISO/IEC/IEEE 29148:2018 \cite{ISO29148}.
\subsection{Purpose} Agreed requirements baseline between Team Fullstack and the city.
\subsection{Scope}
The system shall let residents submit and track geolocated service requests and let
authorized staff triage, assign, update, and close them, with resident status
notifications. Field-crew mobile apps and payment processing are out of scope.
\subsection{Product Overview}
\subsubsection{Product Perspective}
A three-tier web application (browser frontend, API backend, database) integrating
with the city SSO (OIDC), a geocoding/mapping API, and an SMTP relay.
\subsubsection{Product Functions}
Request submission with location/photo; resident status tracking; staff triage,
assignment, and status update; duplicate detection; email notifications; reporting.
\subsubsection{User Characteristics}
Residents (unauthenticated or lightly registered), staff agents, and staff
administrators (role-based). \subsubsection{Assumptions \& Dependencies}
City SSO test realm; provided categories/routing; free-tier geocoding; SMTP relay.

\section{External Interface Requirements}
\subsection{EI-01: City SSO (OIDC)}
\req{Staff authentication shall be delegated to the city identity provider via
OpenID Connect authorization-code flow.}
{Staff must use existing city credentials; the app must not store staff passwords.}
{Sponsor IT.}{Critical}{Test --- staff sign-in via the SSO realm.}
{Depends on the SSO test realm.}{OpenID Connect Core 1.0, OAuth 2.0 (RFC 6749).}
\subsection{EI-02: Geocoding/Mapping API}
\req{The system shall geocode addresses and render request locations on a map via
an external mapping API.}
{Locations must be normalized and visualized for triage.}
{Team.}{High}{Demonstration --- submit an address; see it mapped.}
{Free-tier rate limits.}{None.}

\section{Functional Requirements}
\subsection{FR-01: Submit request}
\req{A resident shall submit a request with category, description, location (map pin
or address), and optional photo, and receive a tracking id.}
{Intake is the entry point of the whole workflow.}
{Sponsor.}{Critical}{Test --- submit and receive a tracking id.}
{Photo size limited; location required.}{None.}
\subsection{FR-02: Track request}
\req{A resident shall view the current status and history of a request by its
tracking id without authentication.}
{Self-service status is the main resident benefit.}
{Sponsor.}{Critical}{Demonstration --- look up a tracking id.}
{No PII of other residents exposed.}{None.}
\subsection{FR-03: Staff triage \& assignment}
\req{An authenticated agent shall filter/sort the request queue, set priority, and
assign a request to a crew or department.}
{Staff need to organize and route the backlog.}
{Sponsor.}{Critical}{Demonstration --- triage and assign a request.}
{Restricted to the agent role (SE-01).}{None.}
\subsection{FR-04: Status update \& notification}
\req{When an agent changes a request status, the system shall record the change and
email the resident (if contact provided).}
{Closes the loop and reduces status-check calls.}
{Sponsor.}{High}{Test --- change status; verify record + email.}
{Requires SMTP relay.}{None.}
\subsection{FR-05: Duplicate detection}
\req{On submission the system shall flag likely duplicates (same category within a
configurable radius/time) for staff review.}
{Cuts duplicate handling effort.}
{Sponsor.}{Moderate}{Test --- near-duplicate submissions are flagged.}
{Radius/time configurable.}{None.}

\section{Usability Requirements}
\subsection{US-01: Accessible, responsive UI}
\req{The resident portal and staff dashboard shall conform to WCAG 2.2 AA and be
usable on mobile and desktop viewport widths (360--1920\,px).}
{Public services must be accessible to all residents on any device.}
{Sponsor; law.}{High}{Test --- automated + manual accessibility audit.}
{Modern browsers.}{WCAG 2.2 AA.}

\section{Performance Requirements}
\subsection{PE-01: Responsiveness}
\req{Interactive pages shall respond within 500\,ms (p95) and support at least 200
concurrent users on the staging host.}
{Sluggish public services are abandoned.}
{Sponsor.}{High}{Test --- load test at target concurrency.}{Bounded by host.}{None.}

\section{Security \& Privacy Requirements}
\subsection{SE-01: RBAC \& transport security}
\req{All staff functions shall be gated by role, and all traffic shall use TLS 1.3;
plaintext HTTP shall redirect to HTTPS.}
{Prevents unauthorized access and eavesdropping on resident data.}
{Sponsor; security review.}{Critical}{Test --- role checks; TLS scan.}
{On staging and production.}{OWASP ASVS; RFC 8446 (TLS 1.3).}
\subsection{SE-02: Resident data minimization}
\req{The system shall collect only the resident contact data needed for
notifications and shall not expose personal data on public tracking pages.}
{Privacy obligation under the city policy.}
{Sponsor policy.}{High}{Inspection --- data-flow and page review.}
{City privacy policy.}{None.}

\section{Maintenance \& Support Requirements}
\subsection{MS-01: Observability}
\req{The system shall emit structured logs and health metrics sufficient to
diagnose failures and monitor uptime without code changes.}
{Operable by city IT after handover.}
{Team.}{Moderate}{Inspection.}{Log retention sized for staging.}
{ISO/IEC 25010:2023 (analysability) \cite{ISO25010}.}

\section{Other Quality Attributes}
\subsection{QA-01: Availability}
\req{The service shall achieve $\geq$99.5\% monthly availability at the public
endpoint.}
{Residents rely on it being reachable.}
{Sponsor.}{High}{Analysis --- uptime measurement.}{Single-region staging.}
{ISO/IEC 25010:2023 (availability) \cite{ISO25010}.}
